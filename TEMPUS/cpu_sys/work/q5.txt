foreach sp $startpoints {
    # Check if it's a port
    if {[llength [get_ports $sp]] > 0} {
        puts " - $sp : INPUT PORT"
    } else {
        set cell [get_cells -of $sp]
        if {[llength $cell] > 0} {
            # Get 'is_sequential' for just one cell
            set is_ff [get_attribute [lindex $cell 0] is_sequential]
            if {$is_ff eq "true"} {
                puts " - $sp : FF OUTPUT"
            } else {
                puts " - $sp : COMBINATIONAL CELL OUTPUT"
            }
        } else {
            puts " - $sp : UNKNOWN TYPE"
        }
    }
}
